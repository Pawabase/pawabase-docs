Every client option that touches the network goes through `fetch`, so a fake `fetch` is a complete fake server.
A small helper that answers by method and path keeps tests short:

<CodeGroup>
```typescript TypeScript
// test/fake-pawabase.ts
import { createClient, type PawabaseClient } from "@pawabase/client";

type Handler = (request: { url: URL; method: string; body: unknown; headers: Headers }) => { status?: number; body?: unknown; headers?: Record<string, string> };

export function fakePawabase(routes: Record<string, Handler>) {
  const requests: Array<{ method: string; url: URL; body: unknown; headers: Headers }> = [];
  const client: PawabaseClient = createClient({
    url: "https://api.example.test",
    apiKey: "pb_pk_test",
    environment: "test",
    retry: false,
    fetch: async (input, init) => {
      const url = new URL(input);
      const method = (init?.method ?? "GET").toUpperCase();
      const headers = new Headers(init?.headers);
      const raw = init?.body;
      const body = typeof raw === "string" ? safeJson(raw) : raw;
      requests.push({ method, url, body, headers });
      const handler = routes[`${method} ${url.pathname}`];
      if (!handler) return new Response("Not found", { status: 404 });
      const answer = handler({ url, method, body, headers });
      return new Response(answer.body === undefined ? null : JSON.stringify(answer.body), {
        status: answer.status ?? 200,
        headers: { "content-type": "application/json", ...answer.headers },
      });
    },
  });
  return { client, requests, done: () => client.dispose() };
}

function safeJson(text: string) {
  try {
    return JSON.parse(text);
  } catch {
    return text;
  }
}

export const page = <T>(data: T[], total: number | null = data.length) => ({ body: { data, page: 1, per_page: 20, total } });
```

```python Python
# tests/fake_pawabase.py
import json

import httpx

from pawabase import Pawabase


class Wire:
    """A fake gateway: records requests and answers from a route table."""

    def __init__(self, routes: dict):
        self.routes, self.requests = routes, []

    def __call__(self, request: httpx.Request) -> httpx.Response:
        self.requests.append(request)
        handler = self.routes.get(f"{request.method} {request.url.path}")
        if handler is None:
            return httpx.Response(404, text="Not found")
        status, body = handler(request)
        return httpx.Response(status, json=body)


def fake_pawabase(routes: dict) -> tuple[Pawabase, Wire]:
    wire = Wire(routes)
    pb = Pawabase("http://gateway.test", "pb_pk_test", environment="test", retries=0)
    # The package's own tests swap the private httpx client for one with a MockTransport.
    pb._client = httpx.Client(transport=httpx.MockTransport(wire), base_url="http://gateway.test", headers=pb.headers)
    return pb, wire


def page(rows, total=None):
    return 200, {"data": rows, "page": 1, "per_page": 20, "total": len(rows) if total is None else total}
```
</CodeGroup>

A test of a service function then reads like a specification:

<CodeGroup>
```typescript TypeScript
import { test } from "node:test";
import assert from "node:assert/strict";
import { fakePawabase, page } from "./fake-pawabase";
import { ordersService } from "../src/orders/service";

test("open orders are requested newest first", async () => {
  const fake = fakePawabase({ "GET /rest/v1/orders": () => page([{ id: "o1", status: "open" }]) });
  const orders = ordersService(fake.client);

  const result = await orders.open();

  assert.equal(result.data.length, 1);
  const params = fake.requests[0].url.searchParams;
  assert.equal(params.get("filter[status]"), "eq.open");
  assert.equal(params.get("sort"), "-created_at");
  fake.done();
});

test("a taken email becomes a field error", async () => {
  const fake = fakePawabase({
    "POST /rest/v1/customers": () => ({ status: 409, body: { error: "unique_violation", message: "taken", details: { columns: ["email"] } } }),
  });
  const result = await signUpCustomer(fake.client, { email: "a@b.c", name: "A" });
  assert.deepEqual(result, { ok: false, fields: { email: ["An account with this email exists."] } });
  fake.done();
});

declare function signUpCustomer(client: unknown, input: { email: string; name: string }): Promise<unknown>;
```

```python Python
import pytest

from pawabase import ConflictError
from tests.fake_pawabase import fake_pawabase, page


def test_open_orders_are_requested_newest_first():
    pb, wire = fake_pawabase({"GET /rest/v1/orders": lambda request: page([{"id": "o1", "status": "open"}])})

    result = pb.from_("orders").eq("status", "open").order("-created_at").execute()

    assert len(result) == 1
    params = wire.requests[0].url.params
    assert params["filter[status]"] == "eq.open"
    assert params["sort"] == "-created_at"


def test_a_taken_email_is_a_conflict():
    pb, _ = fake_pawabase({
        "POST /rest/v1/customers": lambda request: (409, {"error": "unique_violation", "message": "taken", "details": {"columns": ["email"]}}),
    })
    with pytest.raises(ConflictError) as raised:
        pb.from_("customers").create({"email": "a@b.c", "name": "A"})
    assert raised.value.details == {"columns": ["email"]}
```

`retry: false` keeps failures immediate.
Tests of retry behavior itself turn retries back on with tiny delays, as the [errors page](/clients/typescript/errors-retries/testing-error-handling#testing-error-handling) shows.

### Signed-in fakes

To test code that runs as a signed-in person, give the fake client a session first.
`setSession` fetches the user, so the fake answers `/auth/v1/user`:

```typescript
const fake = fakePawabase({
  "GET /auth/v1/user": () => ({ body: { id: "u1", email: "ada@example.com", roles: ["editor"], identities: [], user_metadata: {}, app_metadata: {} } }),
  "GET /rest/v1/tasks": ({ headers }) => (headers.get("authorization") ? page([{ id: "t1" }]) : { status: 401, body: "Authentication required" }),
});

await fake.client.auth.setSession({ access_token: fakeJwt({ exp: Math.floor(Date.now() / 1000) + 900 }), refresh_token: "r1" });

function fakeJwt(claims: Record<string, unknown>) {
  const encode = (value: unknown) => Buffer.from(JSON.stringify(value)).toString("base64url");
  return `${encode({ alg: "none" })}.${encode(claims)}.signature`;
}
```

The client reads the token's expiry from its payload without verifying the signature, so a fake token with an `exp` claim is enough in tests.
Never do this outside tests; the server verifies every real token.

### Time and expiring sessions

Session refresh depends on time.
Test runners that fake timers let a test move past an expiry without waiting:

```typescript
import { test, mock } from "node:test";

test("an expiring session is refreshed before a request", async () => {
  mock.timers.enable({ apis: ["Date", "setTimeout"] });
  const now = Math.floor(Date.now() / 1000);
  let refreshed = 0;
  const fake = fakePawabase({
    "GET /auth/v1/user": () => ({ body: { id: "u1", email: "a@b.c", roles: [], identities: [], user_metadata: {}, app_metadata: {} } }),
    "POST /auth/v1/token": () => {
      refreshed += 1;
      return { body: { access_token: fakeJwt({ exp: now + 2000 }), refresh_token: "r2", expires_in: 900, token_type: "bearer" } };
    },
    "GET /rest/v1/tasks": () => page([]),
  });
  await fake.client.auth.setSession({ access_token: fakeJwt({ exp: now + 900 }), refresh_token: "r1" });

  mock.timers.tick(870 * 1000);
  await fake.client.from("tasks").perPage(20);

  assert.equal(refreshed, 1);
  fake.done();
  mock.timers.reset();
});
```


## Component tests with a mocked network

Component tests render real screens with a network mock such as MSW, which intercepts `fetch` in browsers and in Node.
The client needs no special configuration: it calls `fetch`, and MSW answers.

```typescript
import { http, HttpResponse } from "msw";
import { setupServer } from "msw/node";

const server = setupServer(
  http.get("*/rest/v1/orders", () => HttpResponse.json({ data: [{ id: "o1", status: "open" }], page: 1, per_page: 20, total: 1 })),
  http.post("*/rest/v1/orders/:id/pay", ({ params }) =>
    params.id === "missing" ? HttpResponse.json({ error: "order_not_found", message: "no such order" }, { status: 404 }) : HttpResponse.json({ paid: true }),
  ),
);

beforeAll(() => server.listen());
afterEach(() => server.resetHandlers());
afterAll(() => server.close());

declare function beforeAll(fn: () => void): void;
declare function afterEach(fn: () => void): void;
declare function afterAll(fn: () => void): void;
```

Answer with the platform's real shapes, including error bodies, so components are tested against what they will receive.
Keep typed fixtures aligned with resource definitions so tests use the platform's real shapes.
