```typescript TypeScript
import { createClient } from "@pawabase/client";

const pawabase = createClient({ url: "https://api.example.com", project: "01JC4…", apiKey: "pb_pk_…" });

await pawabase.auth.signInWithPassword({ email: "ada@example.com", password: "…" });

const { data: tasks } = await pawabase.from("tasks").eq("done", false);
```

```python Python
from pawabase import Pawabase

pb = Pawabase("https://api.example.com", "pb_pk_…", project="01JC4…")

pb.auth.sign_in_with_password(email="ada@example.com", password="…")

tasks = pb.from_("tasks").eq("done", False).execute().data
```
