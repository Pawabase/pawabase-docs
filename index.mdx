export const CopyCommand = ({ command }) => {
  const [copied, setCopied] = useState(false);
  return (
    <button type="button" className="pbm-install" onClick={() => { try { navigator.clipboard.writeText(command); setCopied(true); setTimeout(() => setCopied(false), 1600); } catch (e) {} }}>
      <span className="pbm-prompt">$</span>
      <code>{command}</code>
      <span className="pbm-copy">{copied ? "Copied" : "Copy"}</span>
    </button>
  );
};
