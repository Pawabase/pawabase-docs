This quickstart assumes a React app (Vite or similar) and a `tasks` table with `title` and `done` columns. If you do not have one yet, follow [JavaScript](/quickstarts/javascript) steps 1 to 4.

## Install and create the client

```bash
npm install @pawabase/client
```

```typescript
// src/pawabase.ts
import { createClient } from "@pawabase/client";

export const pawabase = createClient({
  url: import.meta.env.VITE_PAWABASE_URL,
  apiKey: import.meta.env.VITE_PAWABASE_KEY,   // a pb_pk_… key
});
```

## Know who is signed in

`onAuthStateChange` tells you when someone signs in or out, and fires once when the page loads with whatever session was stored.

```tsx
// src/useUser.ts
import { useEffect, useState } from "react";
import type { User } from "@pawabase/client";
import { pawabase } from "./pawabase";

export function useUser() {
  const [user, setUser] = useState<User | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const { unsubscribe } = pawabase.auth.onAuthStateChange((_event, session) => {
      setUser(session?.user ?? null);
      setLoading(false);
    });
    return unsubscribe;
  }, []);

  return { user, loading };
}
```

## A sign-in form

```tsx
import { useState } from "react";
import { ValidationError } from "@pawabase/client";
import { pawabase } from "./pawabase";

export function SignIn() {
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [error, setError] = useState("");

  async function submit(e: React.FormEvent) {
    e.preventDefault();
    try {
      const result = await pawabase.auth.signInWithPassword({ email, password });
      if (result.status === "mfa_required") {
        const code = prompt("Enter your authenticator code") ?? "";
        await result.verify(code);
      }
    } catch (err) {
      setError(err instanceof ValidationError ? "Check your details" : "Wrong email or password");
    }
  }

  return (
    <form onSubmit={submit}>
      <input value={email} onChange={(e) => setEmail(e.target.value)} type="email" />
      <input value={password} onChange={(e) => setPassword(e.target.value)} type="password" />
      <button>Sign in</button>
      {error && <p>{error}</p>}
    </form>
  );
}
```

You do not store the result anywhere: the client keeps the session and `useUser` updates by itself.

## A live task list

Load the tasks, then listen for changes on the `tasks` table. This needs **Realtime** switched on for the table in the dashboard.

```tsx
import { useEffect, useState } from "react";
import { pawabase } from "./pawabase";

type Task = { id: string; title: string; done: boolean };

export function Tasks() {
  const [tasks, setTasks] = useState<Task[]>([]);

  useEffect(() => {
    pawabase.from("tasks").order("created_at", "desc").then((page) => setTasks(page.data as Task[]));

    const channel = pawabase.realtime.channel("resource:tasks");
    channel.on("message", ({ event, payload }: any) => {
      const row = payload.record as Task;
      setTasks((current) =>
        event === "deleted" ? current.filter((t) => t.id !== row.id)
        : current.some((t) => t.id === row.id) ? current.map((t) => (t.id === row.id ? row : t))
        : [row, ...current],
      );
    });
    channel.subscribe();
    return () => { channel.unsubscribe(); };
  }, []);

  async function add(title: string) {
    await pawabase.from("tasks").create({ title, done: false });
  }

  return (
    <>
      <button onClick={() => add("New task")}>Add</button>
      <ul>{tasks.map((t) => <li key={t.id}>{t.title}</li>)}</ul>
    </>
  );
}
```

Add a task in one tab and it appears in another, and each person only receives rows they are allowed to read.
