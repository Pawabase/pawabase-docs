Everything here is plain `fetch`. If you are on JavaScript or TypeScript and would rather not write this yourself, the official `@pawabase/client` package already does all of it — see [JavaScript and TypeScript](/clients/javascript/overview). The code below is the reference implementation, and is what you need on a platform without that package.

### Storing the tokens

The tradeoff, stated honestly:

| Store | Survives reload | Readable by injected script | Verdict |
| --- | --- | --- | --- |
| Module variable | No | No | Best, if you can re-authenticate on reload |
| `sessionStorage` | Per tab | Yes | Reasonable middle ground |
| `localStorage` | Yes, shared across tabs | Yes | Common, weakest against XSS |
| httpOnly cookie set by *your* server | Yes | **No** | Best, if you have a server |

If you have no backend of your own, the honest options are memory or `localStorage`, and the difference is real: **anything in `localStorage` is readable by any script running on your page**, including a successful XSS payload, and a stolen refresh token is valid for `refresh_ttl` — 30 days by default — not for the next hour.

```text
const SKEW_SECONDS = 60;
const STORAGE_KEY = "pawabase.session";

export class PawabaseAuth {
  constructor({ baseUrl, apiKey }) {
    this.baseUrl = baseUrl.replace(/\/$/, "");
    this.apiKey = apiKey;
    this.session = this.#load();
    this.#inflight = null;   // the single-flight promise
  }

  #load() {
    const raw = sessionStorage.getItem(STORAGE_KEY);
    return raw ? JSON.parse(raw) : null;
  }

  #save(session) {
    this.session = session;
    if (session) sessionStorage.setItem(STORAGE_KEY, JSON.stringify(session));
    else sessionStorage.removeItem(STORAGE_KEY);
    return session;
  }

  #expired() {
    if (!this.session?.access_token) return true;
    return this.session.expires_at - Math.floor(Date.now() / 1000) <= SKEW_SECONDS;
  }

  async #post(body) {
    const response = await fetch(`${this.baseUrl}/auth/v1/token`, {
      method: "POST",
      headers: { apikey: this.apiKey, "content-type": "application/json" },
      body: JSON.stringify(body),
    });
    const payload = await response.json().catch(() => null);
    if (!response.ok) {
      const error = new Error(payload?.detail ?? `auth failed (${response.status})`);
      error.status = response.status;
      error.detail = payload?.detail ?? null;
      throw error;
    }
    return payload;
  }

  async signIn({ email, password, org }) {
    // `org` is omitted entirely when there is none — sending null is different.
    const body = { grant_type: "password", email, password };
    if (org) body.org = org;
    const result = await this.#post(body);
    if (result.mfa_required) return result;      // caller must complete MFA
    return this.#save(result);
  }

  async completeMfa({ mfaToken, code }) {
    return this.#save(
      await this.#post({ grant_type: "mfa", mfa_token: mfaToken, code }),
    );
  }

  async #refresh() {
    if (!this.session?.refresh_token) throw new Error("no session");
    return this.#post({
      grant_type: "refresh_token",
      refresh_token: this.session.refresh_token,
    });
  }

  /**
   * Single-flight refresh. Concurrent callers share one in-flight request, so
   * the refresh token is only ever presented once.
   */
  async refresh({ force = false } = {}) {
    if (!force && !this.#expired()) return this.session;
    if (!this.#inflight) {
      this.#inflight = this.#refresh()
        .then((next) => this.#save(next))
        .catch((error) => {
          // 401 here means the family is gone: revoked, or reuse was detected.
          if (error.status === 401) this.clear();
          throw error;
        })
        .finally(() => { this.#inflight = null; });
    }
    return this.#inflight;
  }

  async fetch(path, { method = "GET", body, headers = {}, retry = true } = {}) {
    await this.refresh();
    const token = this.session?.access_token;
    const response = await fetch(`${this.baseUrl}${path}`, {
      method,
      headers: {
        apikey: this.apiKey,
        ...(token ? { authorization: `Bearer ${token}` } : {}),
        ...(body ? { "content-type": "application/json" } : {}),
        ...headers,
      },
      body: body ? JSON.stringify(body) : undefined,
    });

    if (response.status === 401 && retry && this.session?.refresh_token) {
      await this.refresh({ force: true });
      return this.fetch(path, { method, body, headers, retry: false });
    }
    return response;
  }

  clear() {
    this.session = null;
    sessionStorage.removeItem(STORAGE_KEY);
  }
}
```

<Warning>
`sessionStorage` is per-tab and per-origin, which has a useful side effect: two tabs each hold their own copy of the session, and a refresh in one does not invalidate the other's refresh token. `localStorage` is shared, so two tabs refreshing from one stored token will trigger theft detection. If you use `localStorage`, you **must** implement single-flight refresh with cross-tab coordination, or you will log users out at random.
</Warning>

### Signing out

Sign-out is a separate endpoint, not a grant on `/token`:

```python
class Logout(BaseModel):
    scope: Literal["local", "global", "others"] = "local"
```

```python
@r.post(
    "/logout",
    request_model=Logout,
    summary="Sign out this session, all sessions, or all others",
)
async def logout(ctx: HttpContext, body: Logout):
    user, principal = await signed_in_user(ctx)
    current = principal.claims.get("sid")
    if body.scope == "global":
        await revoke_all(user)
    elif body.scope == "others":
        from app.sessions import list_sessions

        for session in await list_sessions(user):
            if session["id"] != current:
                await revoke_session(user, session["id"])
    elif current:
        await revoke_session(user, current)
    await log_event(user.project, user.env, "sign_out", user=user, method=body.scope, ctx=ctx)
    await akountz.emit(
        user.project,
        user.env,
        "session.revoked",
        {"user_id": str(user.id), "scope": body.scope},
        actor=str(user.id),
    )
    return {"signed_out": True, "scope": body.scope}
```

| Scope | Effect |
| --- | --- |
| `local` (default) | Revokes this session only |
| `others` | Revokes every session except this one |
| `global` | Revokes all sessions, including this one |

```text
async signOut(scope = "local") {
  try {
    await fetch(`${this.baseUrl}/auth/v1/logout`, {
      method: "POST",
      headers: {
        apikey: this.apiKey,
        authorization: `Bearer ${this.session.access_token}`,
        "content-type": "application/json",
      },
      body: JSON.stringify({ scope }),
    });
  } finally {
    // Always clear locally, even if the network call failed.
    this.clear();
  }
}
```

```bash
curl -X POST "$GW/auth/v1/logout" \
  -H "apikey: $PUBLISHABLE_KEY" \
  -H "Authorization: Bearer $ACCESS_TOKEN" \
  -H "content-type: application/json" \
  -d '{"scope": "others"}'
```

<Info>
`signOut` needs a **valid access token** — the route calls `signed_in_user`, which refuses with `401 sign in first` without one. If the access token has already expired, sign-out will fail and the session stays alive on the server. In that case, clear locally and let the refresh token lapse, or call it before the token expires.
</Info>

### Reading the current user

```python
@r.get("/user", summary="The signed-in user")
async def me(ctx: HttpContext):
    user, principal = await signed_in_user(ctx)
    return {
        **await user_view(user),
        "aal": principal.claims.get("aal"),
        "session_id": principal.claims.get("sid"),
    }
```

```bash
curl "$GW/auth/v1/user" \
  -H "apikey: $PUBLISHABLE_KEY" \
  -H "Authorization: Bearer $ACCESS_TOKEN"
```

Two fields come from the token rather than the database: `aal` and `session_id`. `session_id` is the family id — use it to identify "this session" when you later call logout with `scope: "others"`.

Note what this endpoint is *not*: it does not report the user's organizations, permissions as resolved at evaluation time, or session list. `user_view` includes `roles` and `mfa_enabled`; `permissions` appears only on the admin view, gated behind `admin=True`.
