Because a duplicate refresh revokes the family, this is not an optimisation — it is a correctness requirement for any client that can issue requests in parallel.

### In one tab

The `#inflight` promise in the class above is the whole mechanism. Ten components hitting `401` at once produce **one** network refresh, because the first caller sets `#inflight` and the other nine `await` the same promise.

### Across tabs

`localStorage` is shared, so tabs race. The cheapest correct fix is a storage lock with a short expiry:

```text
const LOCK_KEY = "pawabase.refresh.lock";
const LOCK_TTL_MS = 10_000;

const waitFor = (predicate, timeoutMs) =>
  new Promise((resolve, reject) => {
    const started = Date.now();
    const tick = () => {
      if (predicate()) return resolve();
      if (Date.now() - started > timeoutMs) return reject(new Error("refresh lock timeout"));
      setTimeout(tick, 50);
    };
    tick();
  });

async function withRefreshLock(fn) {
  const now = Date.now();
  const held = JSON.parse(localStorage.getItem(LOCK_KEY) || "null");

  if (held && now < held.until) {
    // Another tab is refreshing. Wait for it to publish a new session.
    await waitFor(() => localStorage.getItem(STORAGE_KEY) !== held.token, 5000);
    return JSON.parse(localStorage.getItem(STORAGE_KEY));
  }

  const lock = { token: localStorage.getItem(STORAGE_KEY), until: now + LOCK_TTL_MS };
  localStorage.setItem(LOCK_KEY, JSON.stringify(lock));
  try {
    return await fn();
  } finally {
    localStorage.removeItem(LOCK_KEY);
  }
}
```

<Info>
Storage locks are advisory — they work because every tab in your app runs this same code. They do not protect against a genuinely malicious script, which would not bother taking the lock. That is precisely why an httpOnly cookie held by your own server is the stronger answer where you have one.
</Info>

### On a server

A server acting on behalf of one user has the same hazard and the same fix: one refresh per user at a time, keyed by user id, with everyone else awaiting the result.

```python
import asyncio
import time
import httpx

REFRESH_LOCKS: dict[str, asyncio.Lock] = {}
SKEW_SECONDS = 60


def _lock_for(user_id: str) -> asyncio.Lock:
    lock = REFRESH_LOCKS.get(user_id)
    if lock is None:
        lock = REFRESH_LOCKS[user_id] = asyncio.Lock()
    return lock


async def get_access_token(user_id: str, session: dict, client: httpx.AsyncClient) -> str:
    """Return a live access token, refreshing at most once per user."""
    async def live() -> bool:
        return session.get("expires_at", 0) - int(time.time()) > SKEW_SECONDS

    if await live():
        return session["access_token"]

    async with _lock_for(user_id):
        # Re-check inside the lock: another coroutine may have refreshed
        # while we waited, and the old refresh token is now consumed.
        if await live():
            return session["access_token"]
        response = await client.post(
            f"{client.base_url}/auth/v1/token",
            headers={"apikey": client.api_key},
            json={"grant_type": "refresh_token", "refresh_token": session["refresh_token"]},
        )
        response.raise_for_status()
        session.update(response.json())
        return session["access_token"]
```

<Warning>
That re-check inside the lock is the part people leave out. Without it, two coroutines queue on the lock, the first refreshes and rotates the token, and the second — using a stale in-memory `session` — presents the consumed refresh token and triggers theft detection against itself.
</Warning>


## On a server with `httpx`

A server-to-server client uses the **secret** key and usually no user token at all. If it does act for a specific user, it forwards that user's access token rather than minting one.

```python
import time
import httpx
from typing import Any


class PawabaseError(Exception):
    def __init__(self, status: int, detail: Any):
        super().__init__(f"Pawabase returned {status}: {detail}")
        self.status = status
        self.detail = detail


def _detail_of(response: httpx.Response) -> Any:
    """Akountz returns the detail as the whole JSON body."""
    try:
        return response.json()
    except ValueError:
        return response.text


class Pawabase:
    """A small, hand-written client. No SDK required."""

    def __init__(self, base_url: str, api_key: str, *, timeout: float = 10.0):
        self.base_url = base_url.rstrip("/")
        self.api_key = api_key
        self._http = httpx.Client(timeout=timeout)

    def close(self) -> None:
        self._http.close()

    def __enter__(self) -> "Pawabase":
        return self

    def __exit__(self, *exc: object) -> None:
        self.close()

    def request(
        self,
        method: str,
        path: str,
        *,
        json: dict | None = None,
        token: str | None = None,
        params: dict | None = None,
    ) -> httpx.Response:
        headers = {"apikey": self.api_key}
        if token:
            headers["authorization"] = f"Bearer {token}"
        response = self._http.request(
            method,
            f"{self.base_url}{path}",
            headers=headers,
            json=json,
            params=params,
        )
        if response.status_code >= 400:
            raise PawabaseError(response.status_code, _detail_of(response))
        return response

    def token(self, **body: Any) -> dict:
        """Any grant on POST /auth/v1/token."""
        body.setdefault("grant_type", "password")
        return self.request("POST", "/auth/v1/token", json=body).json()

    def sign_in(self, email: str, password: str) -> dict:
        return self.token(grant_type="password", email=email, password=password)

    def refresh(self, refresh_token: str, *, org: str | None = None) -> dict:
        body: dict[str, Any] = {
            "grant_type": "refresh_token",
            "refresh_token": refresh_token,
        }
        # Only add `org` when switching. Sending null clears the selection.
        if org is not None:
            body["org"] = org
        return self.token(**body)

    def me(self, access_token: str) -> dict:
        return self.request("GET", "/auth/v1/user", token=access_token).json()

    def sign_out(self, access_token: str, scope: str = "local") -> dict:
        return self.request(
            "POST", "/auth/v1/logout", json={"scope": scope}, token=access_token
        ).json()
```

Usage:

```python
with Pawabase("https://api.yourapp.com", os.environ["PAWABASE_SECRET_KEY"]) as pb:
    session = pb.sign_in("ada@example.com", os.environ["ADA_PASSWORD"])
    profile = pb.me(session["access_token"])
    pb.sign_out(session["access_token"], scope="global")
```

The same class in async form, which is what you want inside an application server because it shares one connection pool and one timeout policy:

```python
class AsyncPawabase:
    def __init__(self, base_url: str, api_key: str, *, timeout: float = 10.0):
        self.base_url = base_url.rstrip("/")
        self.api_key = api_key
        self._http = httpx.AsyncClient(timeout=timeout)

    async def aclose(self) -> None:
        await self._http.aclose()

    async def __aenter__(self) -> "AsyncPawabase":
        return self

    async def __aexit__(self, *exc: object) -> None:
        await self.aclose()

    async def request(
        self,
        method: str,
        path: str,
        *,
        json: dict | None = None,
        token: str | None = None,
    ) -> httpx.Response:
        headers = {"apikey": self.api_key}
        if token:
            headers["authorization"] = f"Bearer {token}"
        response = await self._http.request(
            method, f"{self.base_url}{path}", headers=headers, json=json
        )
        if response.status_code >= 400:
            raise PawabaseError(response.status_code, _detail_of(response))
        return response

    async def refresh(self, refresh_token: str) -> dict:
        response = await self.request(
            "POST",
            "/auth/v1/token",
            json={"grant_type": "refresh_token", "refresh_token": refresh_token},
        )
        return response.json()
```

<Note>
`requests` works identically if that is what your codebase uses — `session.post(url, headers=…, json=…)` maps one-to-one onto `httpx`. Nothing in Pawabase's auth surface depends on the HTTP client.
</Note>
